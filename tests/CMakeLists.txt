cmake_minimum_required(VERSION 3.20)
project(btc_puzzle_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TEST_32BIT_LIMBS "Use portable 32-bit Mbed TLS limbs" OFF)
if(TEST_32BIT_LIMBS)
  add_compile_definitions(MBEDTLS_USER_CONFIG_FILE="${CMAKE_CURRENT_SOURCE_DIR}/mbedtls32_config.h")
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
if(MBEDTLS_SOURCE_DIR)
  add_subdirectory("${MBEDTLS_SOURCE_DIR}" mbedtls EXCLUDE_FROM_ALL)
else()
  include(FetchContent)
  FetchContent_Declare(mbedtls
    URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.6/mbedtls-3.6.6.tar.bz2
    URL_HASH SHA256=8fb65fae8dcae5840f793c0a334860a411f884cc537ea290ce1c52bb64ca007a)
  FetchContent_MakeAvailable(mbedtls)
endif()
find_package(OpenSSL REQUIRED)
add_executable(core_tests core_tests.cpp ../search_core.cpp ../checkpoint_store.cpp ../found_records.cpp ../ripemd160.cpp)
target_include_directories(core_tests PRIVATE ..)
target_link_libraries(core_tests PRIVATE mbedcrypto OpenSSL::Crypto)
target_compile_options(core_tests PRIVATE -UNDEBUG)
option(SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
if(SANITIZERS)
  target_compile_options(core_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(core_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME shared_core COMMAND core_tests)
# Compile the actual sketch command/state layer against small host platform adapters.
if(NOT ARDUINOJSON_INCLUDE_DIR)
  include(FetchContent)
  FetchContent_Declare(arduinojson
    URL https://codeload.github.com/bblanchon/ArduinoJson/tar.gz/refs/tags/v7.4.3
    URL_HASH SHA256=a2a8a277fc3cd0404499199bf1468682ea3688f354be0eaf6af8648366911257)
  FetchContent_MakeAvailable(arduinojson)
  set(ARDUINOJSON_INCLUDE_DIR "${arduinojson_SOURCE_DIR}/src")
endif()
add_executable(firmware_tests firmware_tests.cpp ../search_core.cpp ../checkpoint_store.cpp ../found_records.cpp ../ripemd160.cpp)
target_include_directories(firmware_tests PRIVATE fakes .. "${ARDUINOJSON_INCLUDE_DIR}")
target_link_libraries(firmware_tests PRIVATE mbedcrypto)
target_compile_options(firmware_tests PRIVATE -UNDEBUG)
if(SANITIZERS)
  target_compile_options(firmware_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(firmware_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME command_integration COMMAND firmware_tests)
